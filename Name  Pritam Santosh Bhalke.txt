1. Write a program to count number of words in a string.

#include <stdio.h>
int main()
{
	char str[100];
	int i, count=0;
	printf("Enter a string:" );
	fgets(str,sizeof(str),stdin);
	for(i=0,str[i]!='\0';i++)
	{
		if (str[i]=='' && str[i+1]!='')
		count++;
		printf("number of words=%d",count);
		return 0;
		
	}
}
